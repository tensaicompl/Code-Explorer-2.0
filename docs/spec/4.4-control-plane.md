# Control plane specification

> **Normative.** Changing this document requires a specification change request
> recorded in `docs/plan/ISSUES.md`, and where a stored format is affected, a bump
> of the relevant version constant.
>
> Mirrors section 4.4 of the implementation plan, which is not committed.

The control plane is PostgreSQL 16.

## Migrations

Migrations live in `crates/pdx-server/migrations/NNNN_<name>.sql`, applied with
`sqlx::migrate!`. `CONTROL_SCHEMA_VERSION` = 1.

## Schema

```sql
CREATE TABLE repos (
  repo_id TEXT PRIMARY KEY, clone_url TEXT NOT NULL UNIQUE, name TEXT NOT NULL,
  host_kind TEXT NOT NULL CHECK (host_kind IN ('gitlab','github','bitbucket','generic')),
  host_project_key TEXT, default_branch TEXT NOT NULL DEFAULT 'main',
  enabled BOOLEAN NOT NULL DEFAULT TRUE, created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE TABLE repo_aliases (
  repo_id TEXT REFERENCES repos ON DELETE CASCADE, clone_url TEXT NOT NULL UNIQUE,
  seen_at TIMESTAMPTZ NOT NULL DEFAULT now(), PRIMARY KEY (repo_id, clone_url)
);
CREATE TABLE repo_refs (
  repo_id TEXT REFERENCES repos ON DELETE CASCADE, ref_name TEXT NOT NULL,
  head_sha TEXT, last_seen_at TIMESTAMPTZ, PRIMARY KEY (repo_id, ref_name)
);
CREATE TABLE segments (
  segment_id BIGSERIAL PRIMARY KEY, repo_id TEXT NOT NULL REFERENCES repos,
  commit_sha TEXT NOT NULL, profile TEXT NOT NULL CHECK (profile IN ('structural','precise')),
  engine_version TEXT NOT NULL, build_key TEXT NOT NULL, storage_key TEXT NOT NULL UNIQUE, size_bytes BIGINT NOT NULL,
  content_sha256 TEXT NOT NULL, built_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  build_ms INTEGER NOT NULL, status TEXT NOT NULL CHECK (status IN ('ready','degraded','failed')),
  UNIQUE (repo_id, commit_sha, profile, build_key)
);
CREATE TABLE manifests (
  manifest_id BIGSERIAL PRIMARY KEY, name TEXT NOT NULL,          -- 'trunk' or 'pr:<repo_id>:<number>'
  version BIGINT NOT NULL, created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  links_storage_key TEXT, layout_storage_key TEXT, estate_storage_key TEXT,
  UNIQUE (name, version)
);
CREATE TABLE manifest_segments (
  manifest_id BIGINT REFERENCES manifests ON DELETE CASCADE, repo_id TEXT NOT NULL,
  segment_id BIGINT NOT NULL REFERENCES segments, PRIMARY KEY (manifest_id, repo_id)
);
CREATE TABLE manifest_current (name TEXT PRIMARY KEY, manifest_id BIGINT NOT NULL REFERENCES manifests);
CREATE TABLE jobs (
  job_id BIGSERIAL PRIMARY KEY, kind TEXT NOT NULL,   -- index|precise|links|layout|estate|analytics|permsync|repo_sync
  repo_id TEXT, commit_sha TEXT, manifest_name TEXT, payload JSONB NOT NULL DEFAULT '{}',
  state TEXT NOT NULL CHECK (state IN ('queued','running','done','failed','cancelled')),
  priority INTEGER NOT NULL DEFAULT 5, attempts INTEGER NOT NULL DEFAULT 0,
  queued_at TIMESTAMPTZ NOT NULL DEFAULT now(), started_at TIMESTAMPTZ, finished_at TIMESTAMPTZ,
  worker_id TEXT, error TEXT, dedupe_key TEXT UNIQUE
);
CREATE INDEX jobs_state ON jobs(state, priority, queued_at);
CREATE TABLE principals (
  principal_id TEXT PRIMARY KEY,        -- 'user:<subject>' | 'token:<id>' | 'group:<name>'
  kind TEXT NOT NULL CHECK (kind IN ('user','group','token')), display_name TEXT,
  email TEXT, idp_subject TEXT, role TEXT NOT NULL CHECK (role IN ('admin','maintainer','reader')) DEFAULT 'reader',
  disabled BOOLEAN NOT NULL DEFAULT FALSE, created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE TABLE group_members (group_id TEXT REFERENCES principals, member_id TEXT REFERENCES principals, PRIMARY KEY (group_id, member_id));
CREATE TABLE repo_permissions (
  repo_id TEXT REFERENCES repos ON DELETE CASCADE, principal_id TEXT REFERENCES principals ON DELETE CASCADE,
  source TEXT NOT NULL CHECK (source IN ('host_sync','manual')), synced_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  PRIMARY KEY (repo_id, principal_id)
);
CREATE TABLE api_tokens (
  token_id TEXT PRIMARY KEY, principal_id TEXT NOT NULL REFERENCES principals, token_hash TEXT NOT NULL,
  scopes TEXT[] NOT NULL, expires_at TIMESTAMPTZ, last_used_at TIMESTAMPTZ, revoked BOOLEAN NOT NULL DEFAULT FALSE
);
CREATE TABLE audit_log (
  audit_id BIGSERIAL PRIMARY KEY, at TIMESTAMPTZ NOT NULL DEFAULT now(), principal_id TEXT,
  client TEXT, surface TEXT NOT NULL,   -- mcp|api|ui|admin|cli
  action TEXT NOT NULL, manifest_id BIGINT, repo_ids TEXT[], args_hash TEXT,
  result_rows INTEGER, result_bytes INTEGER, latency_ms INTEGER, outcome TEXT NOT NULL
);
CREATE INDEX audit_at ON audit_log(at);
CREATE TABLE metrics_ts (
  scope TEXT NOT NULL, scope_id TEXT NOT NULL, metric TEXT NOT NULL,
  manifest_id BIGINT NOT NULL REFERENCES manifests, value DOUBLE PRECISION NOT NULL,
  PRIMARY KEY (scope, scope_id, metric, manifest_id)
);
CREATE TABLE annotations (               -- 4.8.4 reviewed human/LLM annotations
  annotation_id BIGSERIAL PRIMARY KEY, target_kind TEXT NOT NULL, target_key TEXT NOT NULL,
  field TEXT NOT NULL, value JSONB NOT NULL, origin TEXT NOT NULL CHECK (origin IN ('file','manual','proposal')),
  status TEXT NOT NULL CHECK (status IN ('proposed','approved','rejected')),
  author TEXT NOT NULL, reviewer TEXT, created_at TIMESTAMPTZ NOT NULL DEFAULT now(), reviewed_at TIMESTAMPTZ
);
CREATE TABLE agent_feedback (
  feedback_id BIGSERIAL PRIMARY KEY, at TIMESTAMPTZ NOT NULL DEFAULT now(), principal_id TEXT,
  manifest_id BIGINT, edge_id TEXT, verdict TEXT NOT NULL CHECK (verdict IN ('correct','wrong','missing')), note TEXT
);
CREATE TABLE settings (key TEXT PRIMARY KEY, value JSONB NOT NULL, updated_at TIMESTAMPTZ NOT NULL DEFAULT now());
-- settings holds admin-adjustable runtime values only: pdx_arch_yaml, pr_comments, model_provider, mcp_client_id.
-- Deployment configuration (secrets, endpoints, limits) comes from Appendix F and is never stored here.
```

## Publication rule

A new trunk manifest version is created in one transaction that:

1. inserts `manifests`,
2. inserts all `manifest_segments` rows (copying the previous version's rows and
   replacing those for the repos that changed),
3. updates `manifest_current`.

Readers pin a `manifest_id` for a session.

## Retention rule

Old manifests are retained for `MANIFEST_RETENTION` = 30 versions or 14 days,
whichever is longer, then their unreferenced segments are garbage-collected by the
`gc` job.
